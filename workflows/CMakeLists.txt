cmake_minimum_required(VERSION 3.24)
project(OpenMSWorkflows VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
set(OPENMS4_DEPENDENCY_LOCK_FILE "${CMAKE_CURRENT_LIST_DIR}/../dependencies.lock.json")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/DesktopApplications.cmake")
openms4_find_core()
openms4_find_package(OpenMSCLI)
openms_desktop_find_gui()
foreach(_tool TOPPAS ExecutePipeline)
  openms_desktop_application(${_tool})
endforeach()
set(TOPPAS_MANIFEST_CATEGORY DesktopWorkflow)
openms_desktop_manifest(openms-workflows Workflows TOPPAS ExecutePipeline)
option(OPENMS_DESKTOP_PIPELINE_TESTS "Run pipelines using installed TOPP tools" OFF)
if(BUILD_TESTING AND OPENMS_DESKTOP_PIPELINE_TESTS)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../tests/pipelines" "${CMAKE_CURRENT_BINARY_DIR}/tests")
endif()
