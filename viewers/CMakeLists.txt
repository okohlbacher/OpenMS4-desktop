cmake_minimum_required(VERSION 3.24)
project(OpenMSViewers VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
set(OPENMS4_DEPENDENCY_LOCK_FILE "${CMAKE_CURRENT_LIST_DIR}/../dependencies.lock.json")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/DesktopApplications.cmake")
openms4_find_core()
openms4_find_package(OpenMSCLI)
openms_desktop_find_gui()
foreach(_tool TOPPView ImageCreator INIFileEditor)
  openms_desktop_application(${_tool})
endforeach()
set(TOPPView_MANIFEST_CATEGORY DesktopViewer)
set(INIFileEditor_MANIFEST_CATEGORY DesktopViewer)
openms_desktop_manifest(openms-viewers Misc TOPPView ImageCreator INIFileEditor)
if(BUILD_TESTING)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../tests/imagecreator" "${CMAKE_CURRENT_BINARY_DIR}/tests")
endif()
